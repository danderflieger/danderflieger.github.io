let activeMajor = category; let activeSub = ""; // Start empty so it picks up the first valid sub-assembly for (let i = 0; i < rows.length; i++) { const row = rows[i]; if (!row || row.length === 0) continue; let major = row[0] || ''; let sub = row[1] || ''; let colC = row[2] || ''; let hours = row[3] || ''; let desc = row[4] || ''; // Skip header rows if (major.toLowerCase().includes('major assembly') || (colC.toLowerCase().includes('date') && hours.toLowerCase().includes('time'))) { continue; } // Update Major Assembly if present, otherwise inherit previous if (major) { activeMajor = major.replace(/^\[merged\]\s*/i, '').trim(); } // Update Sub-Assembly if present in Column B. if (sub) { activeSub = sub.replace(/^\[merged\]\s*/i, '').trim(); } else if (!sub && colC && !hours && !desc) { // If Column B is empty, but Column C acts as a section header row activeSub = colC.replace(/^\[merged\]\s*/i, '').trim(); continue; } // If row represents a valid session entry (has date in colC and valid numeric hours) if (colC && hours && !isNaN(parseFloat(hours))) { const hVal = parseFloat(hours); sessionCount++; const finalMajor = activeMajor || category; const finalSub = activeSub || "General Build"; // Falls back to General Build only if truly nothing was specified if (!majorMap.has(finalMajor)) { majorMap.set(finalMajor, new Map()); } const subMap = majorMap.get(finalMajor); if (!subMap.has(finalSub)) { subMap.set(finalSub, { sessions: [], totalHours: 0 }); } const subObj = subMap.get(finalSub); subObj.sessions.push({ date: colC, hours: hVal, desc }); subObj.totalHours += hVal; } }
